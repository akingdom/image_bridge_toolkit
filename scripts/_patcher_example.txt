def _has_vision_capability(url, model):
    \"\"\"Ask Ollama whether this model advertises a vision capability.

    Uses /api/show, which reports a `capabilities` list from Ollama
    0.6.4 onward. Returns (True, None), (False, None), or (None, note)
    when the capability list is unavailable so the caller can fall back
    to a live probe.
    \"\"\"
    try:
        r = requests.post(
            f"{url}/api/show", json={"model": model}, timeout=10,
        )
    except requests.exceptions.RequestException as e:
        return None, f"could not query /api/show: {e}"
    if r.status_code >= 400:
        return None, f"/api/show returned HTTP {r.status_code}"
    try:
        data = r.json()
    except Exception:
        return None, "/api/show returned non-JSON"
    caps = data.get("capabilities")
    if not isinstance(caps, list):
        return None, ("this Ollama version does not report model "
                      "capabilities (0.6.4+ reports them)")
    lowered = [str(c).lower() for c in caps]
    return ("vision" in lowered), None


def _probe_vision_load(url, model, num_ctx=8192):
    \"\"\"Send a 1x1 image and classify the outcome.

    Returns (ok, kind, detail):

        ok     True if the request succeeded.
        kind   "ok" | "oom" | "unsupported" | "error"
        detail Human-readable explanation; the Ollama body on failure.
    \"\"\"
    try:
        r = requests.post(
            f"{url}/api/generate",
            json={
                "model": model,
                "prompt": ("What colour is the single pixel in this "
                           "image? Answer with one colour word."),
                "images": [_PROBE_PNG_B64],
                "stream": False,
                "options": {"num_predict": 6, "num_ctx": num_ctx},
            },
            timeout=120,
        )
    except requests.exceptions.Timeout:
        return False, "error", "request timed out (120s)"
    except requests.exceptions.RequestException as e:
        return False, "error", f"request failed: {e}"

    if r.status_code < 400:
        try:
            text = (r.json().get("response") or "").strip()
        except Exception:
            text = ""
        return True, "ok", (f"answered {text[:60]!r}" if text
                            else "accepted image request")

    body = ""
    try:
        body = r.text[:500]
    except Exception:
        pass
    low = body.lower()

    if any(s in low for s in (
        "does not support image",
        "image input is not supported",
        "does not support vision",
        "unsupported content",
        "images are not supported",
    )):
        return False, "unsupported", body

    if any(s in low for s in (
        "out-of-memory",
        "out of memory",
        "failed to allocate",
        "projector cpu offload",
        "startup failed",
        "llama-server process has terminated",
        "signal: killed",
    )):
        return False, "oom", body

    return False, "error", body

def _p_subject_cluster() -> List[Patch]:
    return [
        Patch(
            "normalise record shape before fingerprint",
            old="""\
    # Fingerprints need ai_desc and visual_sig. Records passed by the
    # pipeline via collect_from are task results, which have ai_desc.
    fps = []
    for i, r in enumerate(records):
        if not isinstance(r, dict):
            continue
        # The collect_from mechanism passes task results. The result of a
        # describe task IS the ai_desc dict plus whatever the machine added,
        # but not the visual_sig — that lives in the cache. So use the
        # fingerprint the machine already computed in the cache if present.
        fp = r.get("subject_fingerprint")
        if fp is None:
            fp = fingerprint({"ai_desc": json.dumps(r),
                              "visual_sig": r.get("visual_sig")})
        if fp is not None:
            key = r.get("__rel_path") or r.get("image_path") or f"item_{i}"
            fps.append((key, fp))""",
            new="""\
    # Records arrive in one of two shapes:
    #   * full cache entries:    {"ai_desc": {...}, "visual_sig": [...]}
    #   * bare ai_desc dicts:    {"sections": {...}, "categories": [...]}
    # Normalise to what fingerprint() expects: a dict with an "ai_desc"
    # key whose value is the description (dict or JSON string), plus an
    # optional "visual_sig". The previous version wrapped the whole
    # record with json.dumps, producing a top-level object with no
    # "categories" and making fingerprint() return None for every
    # record — silently collapsing every cluster to a singleton.
    def _as_entry(r):
        if not isinstance(r, dict):
            return None
        if "ai_desc" in r:
            return r
        if any(k in r for k in ("sections", "categories", "description")):
            return {"ai_desc": r, "visual_sig": r.get("visual_sig")}
        return None

    fps = []
    for i, r in enumerate(records):
        entry = _as_entry(r)
        if entry is None:
            continue
        fp = r.get("subject_fingerprint")
        if fp is None:
            fp = fingerprint(entry)
        if fp is not None:
            key = r.get("__rel_path") or r.get("image_path") or f"item_{i}"
            fps.append((key, fp))""",
        ),
    ]
